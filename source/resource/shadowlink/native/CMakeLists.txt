cmake_minimum_required(VERSION 3.16)
project(shadowlink_native VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
add_library(shadowlink_core SHARED src/protocol.c src/crypto.c src/session.c src/identity.c)
target_include_directories(shadowlink_core PUBLIC src)
target_link_libraries(shadowlink_core PRIVATE OpenSSL::Crypto)
target_compile_options(shadowlink_core PRIVATE -Wall -Wextra -Werror -Wpedantic)
set_target_properties(shadowlink_core PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN")
enable_testing()
add_executable(shadowlink-core-test test/core_test.c)
target_link_libraries(shadowlink-core-test PRIVATE shadowlink_core)
target_compile_options(shadowlink-core-test PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME core COMMAND shadowlink-core-test)

find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0 gio-unix-2.0)
find_path(JSON_C_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSON_C_LIBRARY NAMES json-c libjson-c.so.5 REQUIRED)
find_package(CURL REQUIRED)
add_library(shadowlink_runtime STATIC src/config.c src/log.c src/io.c src/unbind.c src/device.c src/handler.c src/reset-notification.c src/media-sync.c src/media-sync-http.c)
target_include_directories(shadowlink_runtime PUBLIC src ${JSON_C_INCLUDE_DIR})
target_link_libraries(shadowlink_runtime PUBLIC shadowlink_core PkgConfig::GIO ${JSON_C_LIBRARY} CURL::libcurl OpenSSL::Crypto m)
target_compile_options(shadowlink_runtime PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_executable(shadowlink-reset-lock-test test/reset_lock_test.c)
target_link_libraries(shadowlink-reset-lock-test PRIVATE shadowlink_runtime)
target_compile_options(shadowlink-reset-lock-test PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME reset-lock COMMAND shadowlink-reset-lock-test)
add_executable(shadowlink-reset-logging-test test/reset_logging_test.c)
target_link_libraries(shadowlink-reset-logging-test PRIVATE shadowlink_runtime)
target_compile_options(shadowlink-reset-logging-test PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME reset-logging COMMAND shadowlink-reset-logging-test)
add_executable(shadowlink-log-test test/log_test.c)
target_link_libraries(shadowlink-log-test PRIVATE shadowlink_runtime)
target_compile_options(shadowlink-log-test PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME log-redaction COMMAND shadowlink-log-test)
add_executable(shadowlink-backlight-test test/backlight_test.c)
target_link_libraries(shadowlink-backlight-test PRIVATE shadowlink_runtime)
target_compile_options(shadowlink-backlight-test PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME backlight COMMAND shadowlink-backlight-test)
add_executable(shadowlink-server src/main.c src/ble.c)
target_link_libraries(shadowlink-server PRIVATE shadowlink_runtime)
target_compile_options(shadowlink-server PRIVATE -Wall -Wextra -Werror -Wpedantic)
set_target_properties(shadowlink-server PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS shadowlink-server shadowlink_core RUNTIME DESTINATION bin LIBRARY DESTINATION lib)

add_executable(shadowlink-privacy-test test/privacy_test.c)
target_link_libraries(shadowlink-privacy-test PRIVATE shadowlink_runtime)
target_compile_options(shadowlink-privacy-test PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME privacy COMMAND shadowlink-privacy-test)

# Hardware-free process for Python/C handler parity checks; never installed.
add_executable(shadowlink-handler-probe src/main.c src/ble.c test/handler_stubs.c)
target_link_options(shadowlink-handler-probe PRIVATE "-Wl,--wrap=g_timeout_add_seconds")
target_link_libraries(shadowlink-handler-probe PRIVATE shadowlink_runtime)
target_compile_options(shadowlink-handler-probe PRIVATE -Wall -Wextra -Werror -Wpedantic)

    target_link_options(shadowlink-handler-probe PRIVATE "-Wl,--wrap=sl_${name}")
endforeach()

target_compile_options(shadowlink-unbind-test PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME unbind COMMAND shadowlink-unbind-test)
add_executable(shadowlink-unbind-recovery-test test/unbind_recovery_test.c)
target_link_libraries(shadowlink-unbind-recovery-test PRIVATE shadowlink_runtime)
foreach(name run notify_unbind_reset http update_call json_write config_reset)
    target_link_options(shadowlink-unbind-recovery-test PRIVATE "-Wl,--wrap=sl_${name}")
